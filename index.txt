    <script>
        // --- JavaScript for Smooth "Swipe" Transition ---
        const totalSlides = 26; // 0 to 25
        let currentSlide = 0;

        function goToSlide(index) {
            if (index >= 0 && index < totalSlides) {
                currentSlide = index;
                const slidesWrapper = document.getElementById('slides-wrapper');
                // Calculate the vertical scroll position (simulate 'swipe down')
                const scrollY = index * window.innerHeight;
                
                // Animate the scroll
                window.scrollTo({
                    top: scrollY,
                    behavior: 'smooth'
                });
            }
        }

        // This function attempts to keep the user locked to the current slide
        window.addEventListener('scroll', () => {
            const scrollIndex = Math.round(window.scrollY / window.innerHeight);
            if (scrollIndex !== currentSlide) {
                 window.scrollTo({
                    top: currentSlide * window.innerHeight,
                    behavior: 'auto'
                });
            }
        });
        
    </script>
